Developer Quickstart (Python ORM)

Adding dynamic help to any existing or new field requires zero JavaScript coding:

from odoo import fields, models

class SaleOrder(models.Model):
    _inherit = 'sale.order'

    credit_check_helper = fields.Char(
        string="Customer Risk",
        helper="_get_customer_risk_helper",
        helper_size="md", # sm, md, lg, xl, or custom helper_width="450px"
    )

    def _get_customer_risk_helper(self):
        # 'self' has access to in-memory draft values!
        return {
            'Customer': self.partner_id.name or 'Draft',
            'Unpaid Invoices': len(self.partner_id.invoice_ids.filtered(lambda i: i.state == 'posted')),
            'Total Amount': f"{self.amount_total:,.2f} {self.currency_id.symbol}",
        }
